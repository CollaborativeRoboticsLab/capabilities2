Create the workspace folder by running following commands in the terminal.


\begin{DoxyCode}{0}
\DoxyCodeLine{mkdir\ -\/p\ /home/\$USER/capabilities\_ws/src}
\DoxyCodeLine{cd\ /home/\$USER/capabilities\_ws/src}

\end{DoxyCode}

\begin{DoxyCode}{0}
\DoxyCodeLine{git\ clone\ https://github.com/CollaborativeRoboticsLab/capabilities2.git}

\end{DoxyCode}


Optionally you can clone


\begin{DoxyCode}{0}
\DoxyCodeLine{git\ clone\ https://github.com/CollaborativeRoboticsLab/std\_capabilities.git}

\end{DoxyCode}
