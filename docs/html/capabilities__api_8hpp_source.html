73 bool start_capability(rclcpp::Node::SharedPtr node, const std::string& capability, const std::string& provider)
74 {
75 // return value
76 bool value = true;
77
78 // get the running model from the db
79 models::running_model_t running = cap_db_->get_running(provider);
80
81 // start all dependencies
82 // go through the running model and start the necessary dependencies
83 for (const auto& run : running.dependencies)
84 {
85 RCLCPP_INFO(logging_->get_logger(), "found dependency: %s", run.interface.c_str());
86
87 // make an internal 'use' bond for the capability dependency
88 bind_dependency(run.interface);
89
90 // add the runner to the cache
91 value = value and start_capability(node, run.interface, run.provider);
92 }
93
94 // get the provider specification for the capability
95 models::run_config_model_t run_config = cap_db_->get_run_config(provider);
96
97 // create a new runner, this call implicitly starts the runner
98 // create a runner id which is the cap name to uniquely identify the runner
99 // this means only one runner per capability name
100 //
101 // TODO: consider the logic for multiple runners per capability
102 try
103 {
104 runner_cache_.add_runner(node, capability, run_config);
105
106 // log
107 RCLCPP_INFO(logging_->get_logger(), "started capability: %s with provider: %s", capability.c_str(),
108 provider.c_str());
109
110 return value and true;
111 }
112 catch (const capabilities2_runner::runner_exception& e)
113 {
114 RCLCPP_WARN(logging_->get_logger(), "could not start runner: %s", e.what());
115 return false;

125 void stop_capability(const std::string& capability)
126 {
127 // make sure provider exists
128 // this can happen if dependencies fail to resolve in the first place
129 if (!runner_cache_.running(capability))
130 {
131 RCLCPP_ERROR(logging_->get_logger(), "could not get provider for: %s", capability.c_str());
132 return;
133 }
134
135 // get the provider from runner
136 std::string provider = runner_cache_.provider(capability);
137
138 // get the running model from the db
139 models::running_model_t running = cap_db_->get_running(provider);
140
141 // unbind and stop dependencies
142 // FIXME: this unrolls the dependency tree from the bottom up but should probably be top down
143 for (const auto& run : running.dependencies)
144 {
145 RCLCPP_INFO(logging_->get_logger(), "freeing dependency: %s of : %s", run.interface.c_str(), capability.c_str());
146
147 // remove the internal 'use' bond for the capability dependency
148 unbind_dependency(run.interface);
149
150 // stop the dependency if no more bonds
151 if (!bond_cache_.exists(run.interface))
152 {
153 stop_capability(run.interface);
154 }
155 }
156
157 // remove the runner
158 // this will implicitly stop the runner
159 try
160 {
161 runner_cache_.remove_runner(capability);
162 }
163 catch (const capabilities2_runner::runner_exception& e)
164 {
165 RCLCPP_WARN(logging_->get_logger(), "could not stop runner: %s", e.what());
166 return;
167 }
168
169 // log
170 RCLCPP_INFO(logging_->get_logger(), "stopped capability: %s", capability.c_str());
171 }

256 void connect_capability(const std::string& bond_id, const capabilities2_msgs::msg::CapabilityConnection& connection)
257 {
258 // TODO: implement connection storage for non-running capabilities
259 // could also implicitly increment use counts for capabilities with this bond
260 // so that the capabilities are started and then connected
261 // could also increment use either way (running or not), which would allow tracking connections as 'uses'
262 // all of this would require freeing capabilities
263
264 // validate bonds
265 if (!bond_cache_.exists(connection.source.capability, bond_id) ||
266 !bond_cache_.exists(connection.target.capability, bond_id))
267 {
268 RCLCPP_ERROR(logging_->get_logger(), "no bond with id: %s exists for these capabilities", bond_id.c_str());
269 return;
270 }
271
272 // check if source and target capabilities are running
273 if (!runner_cache_.running(connection.source.capability) || !runner_cache_.running(connection.target.capability))
274 {
275 RCLCPP_ERROR(logging_->get_logger(), "both source and target capabilities must be running to connect");
276 return;
277 }
278
279 try
280 {
281 // create a unique connection id for the connection using the bond_id and instance ids
282 //(format: "bond_id/instance_id/target_instance_id")
283 const std::string connection_id = bond_id + '/' + connection.source.instance_id + '/' + connection.target.instance_id;
284
285 // need to pass event emitter to the runner cache so that it can be set in the runner
286 // this allows the runner to emit events on state changes
287 // default runner behaviour does not use event subsystem
288 runner_cache_.add_connection(connection.source.capability, connection_id, connection, event_);
289
290 // log
291 RCLCPP_INFO(logging_->get_logger(), "set 'type %d' connection for runner: %s with id: %s", connection.type.code,
292 connection.source.capability.c_str(), connection_id.c_str());
293 }
294 catch (const capabilities2_runner::runner_exception& e)
295 {
296 RCLCPP_ERROR(logging_->get_logger(), "could not connect runners: %s", e.what());

451 const capabilities2_msgs::msg::CapabilitySpec get_capability_spec(const std::string& resource)
452 {
453 capabilities2_msgs::msg::CapabilitySpec msg;
454
455 // search db for resource
456 // interfaces

458
459 if (!interface.header.name.empty())
460 {
461 // package name is first part of name separated by /

468 msg.content = YAML::Dump(spec);
469 }
470
471 // semantic interfaces

473
474 if (!semantic_interface.header.name.empty())

479 spec["name"] = spec["name"].as<std::string>().substr(semantic_interface.header.name.find_first_of('/') + 1);
480 msg.content = YAML::Dump(spec);

482
483 // providers
484 models::provider_model_t provider = cap_db_->get_provider(resource);
485
486 if (!provider.header.name.empty())

679 // this is a bond without a live external connection
680 // this will help keep the capability runners scoped to a reference

682 // this is specifically needed for running dependencies of a capability
683 void bind_dependency(const std::string& capability)
684 {

696 void unbind_dependency(const std::string& capability)
697 {
698 // do we have a bond id for this capability
699
700 // get cap bonds
701 std::vector<std::string> cap_bonds = bond_cache_.get_bonds(capability);
702
703 // if any internal bonds in cap_bonds
704 for (const auto& bond : cap_bonds)
705 {
706 auto it = std::find(internal_bond_ids_.begin(), internal_bond_ids_.end(), bond);
707 if (it != internal_bond_ids_.end())
708 {
709 // remove just the first bond id found in cap_bonds and internal_bond_ids_
710 bond_cache_.remove_bond(capability, bond);
711
712 // delete internal bond
713 internal_bond_ids_.erase(it);
714
715 return;
716 }
717 }
